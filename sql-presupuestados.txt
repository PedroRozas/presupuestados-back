CREATE TABLE public.budgets (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  owner_id uuid NOT NULL,
  name text NOT NULL,
  type text NOT NULL,
  limit numeric NOT NULL,
  user_id uuid,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  couple_id uuid,
  associated_card text,
  default_split_method text DEFAULT '50/50'::text,
  CONSTRAINT budgets_pkey PRIMARY KEY (id),
  CONSTRAINT budgets_owner_id_fkey FOREIGN KEY (owner_id) REFERENCES auth.users(id),
  CONSTRAINT budgets_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.family_members(id),
  CONSTRAINT budgets_couple_id_fkey FOREIGN KEY (couple_id) REFERENCES public.couples(id)
);
